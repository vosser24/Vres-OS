PERSONAL_TRANSITION_PLAN_SCHEMA = "vres-personal-chairman-transition-plan/1"

_MAX_CLAIMED_BYTES = 512 * 1024
_HEX = frozenset("0123456789abcdef")


class PersonalChairmanTransitionPlanError(ValueError):
    """Refuse ambiguous or invalid offline transition inputs."""


def _sha256(value: bytes) -> str:
    return hashlib.sha256(value).hexdigest()


def _validate_claimed_before(role: str, value: bytes) -> tuple[str, int]:
    """Verify only a caller-supplied Sonnet/high document, not its origin."""
    if type(value) is not bytes or not 0 < len(value) <= _MAX_CLAIMED_BYTES:
        raise PersonalChairmanTransitionPlanError(
            f"{role} before-bytes must be nonempty bounded bytes"
        )
    try:
        baseline = materialize_chairman(value, BASELINE_POLICY)
    except ChairmanPolicyError:
        raise PersonalChairmanTransitionPlanError(
            f"{role} claimed baseline is not a valid Chairman document"
        ) from None
    if baseline.content != value:
        raise PersonalChairmanTransitionPlanError(
            f"{role} claimed baseline must be Sonnet/high"
        )
    return _sha256(value), len(value)


def _check_digest_pair(digest: str | None, size: int | None) -> None:
    if digest is None and size is None:
        return
    if (
        type(digest) is not str
        or len(digest) != 64
        or not set(digest) <= _HEX
        or type(size) is not int
        or not 0 < size <= _MAX_CLAIMED_BYTES
    ):
        raise PersonalChairmanTransitionPlanError(
            "Claimed before-bytes digest/size is invalid"
        )


@dataclass(frozen=True, slots=True)
class PersonalChairmanTransitionPlan:
    """An offline diagnostic only, not an authorization or trusted snapshot."""

    preview: PersonalChairmanPreview
    claimed_managed_sha256: str | None
    claimed_distribution_sha256: str | None
    claimed_managed_bytes: int | None
    claimed_distribution_bytes: int | None

    def __post_init__(self) -> None:
        if type(self.preview) is not PersonalChairmanPreview:
            raise PersonalChairmanTransitionPlanError(
                "An accepted pure Chairman preview is required"
            )
        _check_digest_pair(self.claimed_managed_sha256, self.claimed_managed_bytes)
        _check_digest_pair(
            self.claimed_distribution_sha256, self.claimed_distribution_bytes
        )
        if (self.claimed_managed_sha256 is None) != (
            self.claimed_distribution_sha256 is None
        ):
            raise PersonalChairmanTransitionPlanError(
                "Both claimed target copies are required together"
            )

    def reconciliation_findings(self) -> tuple[str, ...]:
        """Describe caller-byte inconsistencies, never assert host verification."""
        if self.claimed_managed_sha256 is None:
            return ("BEFORE_COPIES_NOT_SUPPLIED",)
        findings: list[str] = []
        if self.claimed_managed_sha256 != self.claimed_distribution_sha256:
            findings.append("MANAGED_DISTRIBUTION_DIVERGENCE")
        expected_sha = _sha256(self.preview.template)
        if (
            self.claimed_managed_sha256 != expected_sha
            or self.claimed_distribution_sha256 != expected_sha
        ):
            findings.append("SOURCE_RELEASE_RECONCILIATION_REQUIRED")
        return tuple(findings)

    def manifest(self) -> dict[str, object]:
        """Fresh non-authoritative manifest; no caller bytes, paths or secrets."""
        preview = self.preview.manifest()
        findings = self.reconciliation_findings()
        if findings == ("BEFORE_COPIES_NOT_SUPPLIED",):
            reconciliation = "NOT_ASSESSED"
        elif findings:
            reconciliation = "REQUIRES_RELEASE_RECONCILIATION"
        else:
            reconciliation = "CALLER_SUPPLIED_SOURCE_MATCH"

        return {
            "schema": PERSONAL_TRANSITION_PLAN_SCHEMA,
            "status": "PREPARED_OFFLINE_ONLY",
            "authority": "NO_F1_AUTHORITY",
            "installation": "NOT_INSTALLED",
            "operator_install_consent": "REQUIRED_SEPARATELY",
            "safe_to_install": False,
            "effective_model": "UNKNOWN",
            "effective_model_observed": False,
            "installed_state_verified": False,
            "claimed_before_provenance": "CALLER_SUPPLIED_UNTRUSTED",
            "release_pointer": "NOT_READ",
            "release_manifest_verified": False,
            "trusted_paths_resolved": False,
            "rollback_snapshot": "NOT_CREATED",
            "logical_target_roles": list(LOGICAL_TARGET_ROLES),
            "source_git_blob_lf_normalized": preview["source_git_blob"],
            "source_sha256_as_supplied": preview["source_sha256"],
            "source_bytes_as_supplied": preview["source_bytes"],
            "source_newline": (
                "CRLF" if b"\r\n" in self.preview.template else "LF"
            ),
            "candidate_sha256": preview["candidate_sha256"],
            "candidate_bytes": preview["candidate_bytes"],
            "unchanged_body_sha256": preview["body_sha256"],
            "baseline_model": preview["baseline_model"],
            "baseline_effort": preview["baseline_effort"],
            "requested_model": preview["requested_model"],
            "requested_effort": preview["requested_effort"],
            "changed_fields": ["model", "effort"],
            "managed_before_sha256_claimed": self.claimed_managed_sha256,
            "distribution_before_sha256_claimed": (
                self.claimed_distribution_sha256
            ),
            "managed_before_bytes_claimed": self.claimed_managed_bytes,
            "distribution_before_bytes_claimed": self.claimed_distribution_bytes,
            "managed_after_sha256_proposed": preview["candidate_sha256"],
            "distribution_after_sha256_proposed": preview["candidate_sha256"],
            "reconciliation": reconciliation,
            "reconciliation_findings": list(findings),
        }

    def __bool__(self) -> bool:
        raise TypeError("An offline PERSONAL plan is never install permission")


def plan_personal_chairman_transition(
    accepted_source: bytes,
    *,
    claimed_managed_before: bytes | None = None,
    claimed_distribution_before: bytes | None = None,
) -> PersonalChairmanTransitionPlan:
    """Plan from accepted SOURCE and optional paired, untrusted baseline bytes.

    Absence of the claims leaves the installed state UNKNOWN. Supplied matching
    claims are still not a physical installed-state observation or permission.
    """
    try:
        preview = preview_personal_chairman(accepted_source)
    except PersonalChairmanPreviewError as exc:
        raise PersonalChairmanTransitionPlanError(str(exc)) from None

    if (claimed_managed_before is None) != (claimed_distribution_before is None):
        raise PersonalChairmanTransitionPlanError(
            "Both claimed target copies are required together"
        )
    if claimed_managed_before is None:
        managed = distribution = None
    else:
        managed = _validate_claimed_before("managed", claimed_managed_before)
        distribution = _validate_claimed_before(
            "distribution", claimed_distribution_before
        )
    return PersonalChairmanTransitionPlan(
        preview=preview,
        claimed_managed_sha256=managed[0] if managed else None,
        claimed_distribution_sha256=distribution[0] if distribution else None,
        claimed_managed_bytes=managed[1] if managed else None,
        claimed_distribution_bytes=distribution[1] if distribution else None,
    )

_MAX_EVENTS = 32
_MAX_COPY_BYTES = 512 * 1024
_HEX = frozenset("0123456789abcdef")
_TERMINAL = frozenset(("ABANDONED", "SIMULATED_COMPLETE", "SIMULATED_ROLLED_BACK", "HALTED_UNCERTAIN"))


class PersonalChairmanRecoveryJournalError(ValueError):
    """A journal, snapshot claim, or hypothetical boundary cannot be trusted."""


def _digest(value: object) -> bool:
    return type(value) is str and len(value) == 64 and set(value) <= _HEX


def _pair(value: object) -> bool:
    return type(value) is tuple and len(value) == 2 and all(_digest(x) for x in value)


@dataclass(frozen=True, slots=True)
class OfflineBackupClaim:
    """ONLY a caller-claimed digest and length; never a saved recovery copy."""

    role: str
    sha256: str
    byte_count: int

    def __post_init__(self) -> None:
        if (
            type(self.role) is not str
            or self.role not in LOGICAL_TARGET_ROLES
            or not _digest(self.sha256)
            or type(self.byte_count) is not int
            or not 0 < self.byte_count <= _MAX_COPY_BYTES
        ):
            raise PersonalChairmanRecoveryJournalError("Invalid offline backup claim")


@dataclass(frozen=True, slots=True)
class OfflineBoundary:
    """One ordered hypothetical boundary with exact claimed before/after pairs."""

    sequence: int
    plan_id: str
    action: str
    before: tuple[str, str]
    after: tuple[str, str]


@dataclass(frozen=True, slots=True)
class PersonalChairmanRecoveryJournal:
    """Pure audit model. It is NOT a trusted write, backup, or consent token."""

    plan_id: str
    source_git_blob: str
    source_sha256: str
    candidate_sha256: str
    before: tuple[str, str] | None
    modeled_release_coherence: str
    plan_reconciliation: str
    backup_claims: tuple[OfflineBackupClaim, OfflineBackupClaim] | None
    blockers: tuple[str, ...]
    entries: tuple[OfflineBoundary, ...] = ()

    def manifest(self) -> dict[str, object]:
        state, current = replay_personal_chairman_recovery(self)
        return {
            "schema": PERSONAL_RECOVERY_JOURNAL_SCHEMA,
            "status": state,
            "simulation_only": True,
            "authority": "NO_F1_AUTHORITY",
            "installation": "NOT_INSTALLED",
            "operator_install_consent": "REQUIRED_SEPARATELY",
            "safe_to_install": False,
            "effective_model": "UNKNOWN",
            "effective_model_observed": False,
            "installed_state_verified": False,
            "release_manifest_verified": False,
            "release_pointer": "NOT_READ",
            "trusted_paths_resolved": False,
            "rollback_snapshot": "NOT_CREATED",
            "backup_provenance": "CALLER_DIGEST_CLAIMS_ONLY",
            "modeled_release_coherence": self.modeled_release_coherence,
            "plan_id": self.plan_id,
            "source_git_blob_lf_normalized": self.source_git_blob,
            "source_sha256_as_supplied": self.source_sha256,
            "candidate_sha256_proposed": self.candidate_sha256,
            "logical_target_roles": list(LOGICAL_TARGET_ROLES),
            "before_sha256_claimed": list(self.before) if self.before else None,
            "modeled_sha256": (
                None if state == "HALTED_UNCERTAIN" else (list(current) if current else None)
            ),
            "modeled_state_is_observation": False,
            "blockers": list(self.blockers),
            "journal_events": len(self.entries),
            "recovery_completed_on_host": False,
            "multi_target_atomicity": "NOT_CLAIMED",
        }

    def __bool__(self) -> bool:
        raise TypeError("An offline recovery journal is never install permission")


def new_personal_chairman_recovery_journal(
    plan: PersonalChairmanTransitionPlan,
    *,
    modeled_release_coherence: str = "UNKNOWN",
    managed_backup: OfflineBackupClaim | None = None,
    distribution_backup: OfflineBackupClaim | None = None,
) -> PersonalChairmanRecoveryJournal:
    """Create a modeled journal only; an ASSUMED release is not verified v2.

    Even consistent caller before-copies are insufficient for real installation.
    Unknown release coherence, plan drift, or missing/mismatched backups block
    ALL simulated forward transitions. No paths or raw copies are retained.
    """
    if type(plan) is not PersonalChairmanTransitionPlan:
        raise PersonalChairmanRecoveryJournalError("Exact PT2-A plan is required")
    if type(modeled_release_coherence) is not str or modeled_release_coherence not in (
        "UNKNOWN", "ASSUMED_COHERENT_FOR_SIMULATION"
    ):
        raise PersonalChairmanRecoveryJournalError("Unrecognized release-coherence claim")
    for claim, role in ((managed_backup, "managed"), (distribution_backup, "distribution")):
        if claim is not None and (type(claim) is not OfflineBackupClaim or claim.role != role):
            raise PersonalChairmanRecoveryJournalError("Backup claim has wrong role or type")

    m = plan.manifest()
    before: tuple[str, str] | None = None
    if m["managed_before_sha256_claimed"] is not None:
        before = (m["managed_before_sha256_claimed"], m["distribution_before_sha256_claimed"])
    blockers: list[str] = []
    if m["reconciliation"] == "NOT_ASSESSED":
        blockers.append("BEFORE_COPIES_NOT_SUPPLIED")
    elif m["reconciliation"] != "CALLER_SUPPLIED_SOURCE_MATCH":
        blockers.append("PT2A_RELEASE_RECONCILIATION_REQUIRED")
    if modeled_release_coherence != "ASSUMED_COHERENT_FOR_SIMULATION":
        blockers.append("RELEASE_V2_COHERENCE_UNKNOWN")
    if managed_backup is None or distribution_backup is None:
        blockers.append("BACKUP_DIGEST_CLAIMS_INCOMPLETE")
    elif before is None or any(
        claim.sha256 != expected_sha or claim.byte_count != expected_size
        for claim, expected_sha, expected_size in (
            (managed_backup, m["managed_before_sha256_claimed"], m["managed_before_bytes_claimed"]),
            (distribution_backup, m["distribution_before_sha256_claimed"], m["distribution_before_bytes_claimed"]),
        )
    ):
        blockers.append("BACKUP_DIGEST_CLAIM_MISMATCH")

    # A digest of explicit PT2-A identities and the assumptions, not a signed issuer.
    fields = (
        PERSONAL_RECOVERY_JOURNAL_SCHEMA,
        m["source_git_blob_lf_normalized"],
        m["source_sha256_as_supplied"],
        m["candidate_sha256"],
        *(before or ("NONE", "NONE")),
        modeled_release_coherence,
        *(str(x) for x in (
            m["managed_before_bytes_claimed"], m["distribution_before_bytes_claimed"]
        )),
        *(f"{c.role}:{c.sha256}:{c.byte_count}" if c else "NONE"
          for c in (managed_backup, distribution_backup)),
    )
    plan_id = hashlib.sha256("\x00".join(fields).encode("ascii")).hexdigest()
    journal = PersonalChairmanRecoveryJournal(
        plan_id=plan_id,
        source_git_blob=m["source_git_blob_lf_normalized"],
        source_sha256=m["source_sha256_as_supplied"],
        candidate_sha256=m["candidate_sha256"],
        before=before,
        modeled_release_coherence=modeled_release_coherence,
        plan_reconciliation=m["reconciliation"],
        backup_claims=(managed_backup, distribution_backup)
        if managed_backup and distribution_backup else None,
        blockers=tuple(blockers),
    )
    replay_personal_chairman_recovery(journal)
    return journal


def replay_personal_chairman_recovery(
    journal: PersonalChairmanRecoveryJournal,
) -> tuple[str, tuple[str, str] | None]:
    """Validate the full hypothetical event chain from its beginning each time."""
    if type(journal) is not PersonalChairmanRecoveryJournal:
        raise PersonalChairmanRecoveryJournalError("Exact offline journal is required")
    if not _digest(journal.plan_id) or not _digest(journal.candidate_sha256):
        raise PersonalChairmanRecoveryJournalError("Invalid bound plan identity")
    if type(journal.entries) is not tuple or len(journal.entries) > _MAX_EVENTS:
        raise PersonalChairmanRecoveryJournalError("Invalid or excessive journal events")
    if journal.before is not None and not _pair(journal.before):
        raise PersonalChairmanRecoveryJournalError("Invalid original target digests")
    if journal.blockers:
        if journal.entries:
            raise PersonalChairmanRecoveryJournalError("Blocked journal cannot advance")
        return "BLOCKED", journal.before
    if journal.before is None or journal.backup_claims is None:
        raise PersonalChairmanRecoveryJournalError("Unbound rollback baseline")
    if journal.modeled_release_coherence != "ASSUMED_COHERENT_FOR_SIMULATION":
        raise PersonalChairmanRecoveryJournalError("Release coherence is not modeled")
    if journal.plan_reconciliation != "CALLER_SUPPLIED_SOURCE_MATCH":
        raise PersonalChairmanRecoveryJournalError("PT2A release reconciliation is unresolved")
    if any(
        claim.sha256 != sha or claim.role != role
        for claim, sha, role in zip(journal.backup_claims, journal.before, LOGICAL_TARGET_ROLES)
    ):
        raise PersonalChairmanRecoveryJournalError("Rollback claim has drifted")
    current = journal.before
    state = "DRAFT"
    for step, entry in enumerate(journal.entries, 1):
        if (
            type(entry) is not OfflineBoundary
            or type(entry.sequence) is not int
            or entry.sequence != step
            or entry.plan_id != journal.plan_id
            or not _pair(entry.before)
            or not _pair(entry.after)
            or entry.before != current
        ):
            raise PersonalChairmanRecoveryJournalError("Journal sequence or before-state mismatch")
        action = entry.action
        if type(action) is not str or state in _TERMINAL:
            raise PersonalChairmanRecoveryJournalError("Illegal event after terminal boundary")
        next_state = None
        after = current
        if action == "STAGE" and state == "DRAFT":
            next_state = "STAGED"
        elif action == "PREPARE" and state == "STAGED":
            next_state = "PREPARED"
        elif action == "WRITE_MANAGED" and state == "PREPARED":
            next_state = "MANAGED_UPDATED"
            after = (journal.candidate_sha256, current[1])
        elif action == "WRITE_DISTRIBUTION" and state == "MANAGED_UPDATED":
            next_state = "BOTH_UPDATED"
            after = (current[0], journal.candidate_sha256)
        elif action == "RECONCILE_AFTER" and state == "BOTH_UPDATED":
            next_state = "SIMULATED_COMPLETE"
        elif action == "ABANDON" and state in ("DRAFT", "STAGED", "PREPARED"):
            next_state = "ABANDONED"
        elif action == "FAIL" and state in ("DRAFT", "STAGED", "PREPARED"):
            next_state = "ABANDONED"
        elif action == "FAIL" and state in ("MANAGED_UPDATED", "BOTH_UPDATED", "RECOVERY_REQUIRED"):
            next_state = "RECOVERY_REQUIRED"
        elif action == "RESTORE_DISTRIBUTION" and state in ("RECOVERY_REQUIRED", "DISTRIBUTION_RESTORED"):
            next_state = "DISTRIBUTION_RESTORED"
            after = (current[0], journal.before[1])
        elif action == "RESTORE_MANAGED" and state in ("DISTRIBUTION_RESTORED", "BOTH_RESTORED"):
            next_state = "BOTH_RESTORED"
            after = (journal.before[0], current[1])
        elif action == "FAIL" and state in ("DISTRIBUTION_RESTORED", "BOTH_RESTORED"):
            next_state = state  # interrupted simulated recovery can be retried
        elif action == "RECONCILE_BEFORE" and state == "BOTH_RESTORED":
            next_state = "SIMULATED_ROLLED_BACK"
        elif action == "UNCERTAIN":
            next_state = "HALTED_UNCERTAIN"  # no invented readable target hash
        if next_state is None or after != entry.after:
            raise PersonalChairmanRecoveryJournalError("Illegal boundary or target digest drift")
        state, current = next_state, after
    return state, current


def append_personal_chairman_recovery_event(
    journal: PersonalChairmanRecoveryJournal,
    event: OfflineBoundary,
) -> PersonalChairmanRecoveryJournal:
    """Append ONE caller-claimed event, revalidating entire immutable history."""
    replay_personal_chairman_recovery(journal)
    if type(event) is not OfflineBoundary:
        raise PersonalChairmanRecoveryJournalError("Exact journal event is required")
    new_journal = replace(journal, entries=journal.entries + (event,))
    replay_personal_chairman_recovery(new_journal)
    return new_journal

def _source(newline: bytes = b"\n") -> bytes:
    result = SOURCE.read_bytes().replace(b"\r\n", b"\n")
    assert b"\r" not in result
    return result.replace(b"\n", newline)


def _plan(source: bytes, before: bytes | None = None):
    return plan_personal_chairman_transition(
        source,
        claimed_managed_before=before,
        claimed_distribution_before=before,
    )


def _ready(newline: bytes = b"\n"):
    original = _source(newline)
    digest = hashlib.sha256(original).hexdigest()
    return new_personal_chairman_recovery_journal(
        _plan(original, original),
        modeled_release_coherence="ASSUMED_COHERENT_FOR_SIMULATION",
        managed_backup=OfflineBackupClaim("managed", digest, len(original)),
        distribution_backup=OfflineBackupClaim("distribution", digest, len(original)),
    )


def _step(journal, action: str, after=None):
    _, before = replay_personal_chairman_recovery(journal)
    if after is None:
        after = before
        if action == "WRITE_MANAGED":
            after = (journal.candidate_sha256, before[1])
        elif action == "WRITE_DISTRIBUTION":
            after = (before[0], journal.candidate_sha256)
        elif action == "RESTORE_DISTRIBUTION":
            after = (before[0], journal.before[1])
        elif action == "RESTORE_MANAGED":
            after = (journal.before[0], before[1])
    event = OfflineBoundary(len(journal.entries) + 1, journal.plan_id, action, before, after)
    return append_personal_chairman_recovery_event(journal, event)


def _forward(journal, stop: str):
    for event in ("STAGE", "PREPARE", "WRITE_MANAGED", "WRITE_DISTRIBUTION"):
        journal = _step(journal, event)
        if event == stop:
            break
    return journal

def test_two_target_happy_simulation_stays_non_authoritative():
    j = _ready()
    for action, expected in (
        ("STAGE", "STAGED"),
        ("PREPARE", "PREPARED"),
        ("WRITE_MANAGED", "MANAGED_UPDATED"),
        ("WRITE_DISTRIBUTION", "BOTH_UPDATED"),
        ("RECONCILE_AFTER", "SIMULATED_COMPLETE"),

        assert j.manifest()["status"] == expected

    assert j.manifest()["installation"] == "NOT_INSTALLED"
    assert j.manifest()["recovery_completed_on_host"] is False

        _step(j, "FAIL")


@pytest.mark.parametrize("stop", ["STAGE", "PREPARE", "WRITE_MANAGED", "WRITE_DISTRIBUTION"])
def test_failure_after_each_forward_boundary_is_deterministic_and_fail_closed(stop):
    j = _forward(_ready(), stop)
    j = _step(j, "FAIL")
    if stop in ("STAGE", "PREPARE"):
        assert j.manifest()["status"] == "ABANDONED"
        assert j.manifest()["modeled_sha256"] == list(j.before)
        with pytest.raises(PersonalChairmanRecoveryJournalError):
            _step(j, "RECONCILE_AFTER")
    else:
        assert j.manifest()["status"] == "RECOVERY_REQUIRED"
        assert j.manifest()["modeled_sha256"][0] == j.candidate_sha256
        with pytest.raises(PersonalChairmanRecoveryJournalError):
            _step(j, "RECONCILE_AFTER")
        j = _step(j, "RESTORE_DISTRIBUTION")
        j = _step(j, "RESTORE_MANAGED")
        j = _step(j, "RECONCILE_BEFORE")
        assert j.manifest()["status"] == "SIMULATED_ROLLED_BACK"
        assert j.manifest()["modeled_sha256"] == list(j.before)

"""PT2-C: pure offline, caller-claimed release-v2 reconciliation decision.

No manifest or installed file is read. A coherent projection of a v2 manifest,
active pointer and logical target digests is NOT release proof or install consent.
The candidate Chairman changes a file governed by the existing v2 manifest;
therefore even consistent claims require a separately designed release artifact.

import hashlib
from dataclasses import dataclass

from .personal_chairman_transition_plan import PersonalChairmanTransitionPlan
from .personal_chairman_recovery_journal import (
    PersonalChairmanRecoveryJournal,

V2_MANIFEST_FORMAT = "vres-personal-release-artifact"
CHAIRMAN_RELEASE_PATH = "plugins/vres-os/agents/chairman.md"
_HEX40 = frozenset("0123456789abcdef")
_MAX_CHAIRMAN_BYTES = 512 * 1024


class PersonalReleaseReconciliationError(ValueError):
    """Refuse a malformed caller-supplied release projection."""


def _hex(value: object, length: int) -> bool:
    return type(value) is str and len(value) == length and set(value) <= _HEX40


def _release_id(value: object) -> bool:
    return (
        type(value) is str and 1 <= len(value) <= 80
        and value[0].isalnum() and value[0].isascii()
        and all(c.isascii() and (c.isalnum() or c in "_-") for c in value)
    )


def _sha256(value: bytes) -> str:
    return hashlib.sha256(value).hexdigest()


def _git_blob(value: bytes) -> str:
    return hashlib.sha1(b"blob " + str(len(value)).encode("ascii") + b"\x00" + value).hexdigest()


@dataclass(frozen=True, slots=True)
class ClaimedV2Manifest:
    """A caller-supplied projection of ONE v2 manifest entry, not the full manifest."""

    format: str
    version: int
    source_origin: str
    object_format: str
    source_commit: str
    source_tree: str
    chairman_path: str
    chairman_mode: str
    chairman_git_blob: str
    chairman_sha256: str
    chairman_bytes: int

    def __post_init__(self) -> None:
        if (
            type(self.format) is not str or len(self.format) > 80
            or type(self.version) is not int or not 0 <= self.version <= 100
            or type(self.source_origin) is not str or len(self.source_origin) > 80
            or type(self.object_format) is not str or len(self.object_format) > 40
            or not _hex(self.source_commit, 40) or not _hex(self.source_tree, 40)
            or type(self.chairman_path) is not str or len(self.chairman_path) > 160
            or type(self.chairman_mode) is not str or len(self.chairman_mode) > 16
            or not _hex(self.chairman_git_blob, 40)
            or not _hex(self.chairman_sha256, 64)
            or type(self.chairman_bytes) is not int
            or not 0 < self.chairman_bytes <= _MAX_CHAIRMAN_BYTES
        ):
            raise PersonalReleaseReconciliationError("Malformed claimed release-v2 entry")


@dataclass(frozen=True, slots=True)
class ClaimedV2Installation:
    """Abstract logical roles only; none of these fields is a host observation."""

    manifest: ClaimedV2Manifest
    release_directory_id: str
    active_pointer_release: str | None
    managed_before_sha256: str
    distribution_before_sha256: str
    pending_install: str = "UNKNOWN"
    plugin_tree_coherence: str = "UNKNOWN"
    reparse_points: str = "UNKNOWN"
    claude_session: str = "UNKNOWN"

    def __post_init__(self) -> None:
        if (
            type(self.manifest) is not ClaimedV2Manifest
            or not _release_id(self.release_directory_id)
            or (self.active_pointer_release is not None
                and not _release_id(self.active_pointer_release))
            or not _hex(self.managed_before_sha256, 64)
            or not _hex(self.distribution_before_sha256, 64)
            or type(self.pending_install) is not str
            or self.pending_install not in ("UNKNOWN", "ABSENT_CLAIMED", "PRESENT_CLAIMED")
            or type(self.plugin_tree_coherence) is not str
            or self.plugin_tree_coherence not in ("UNKNOWN", "MATCH_CLAIMED", "DRIFT_CLAIMED")
            or type(self.reparse_points) is not str
            or self.reparse_points not in ("UNKNOWN", "NONE_CLAIMED", "PRESENT_CLAIMED")
            or type(self.claude_session) is not str
            or self.claude_session not in ("UNKNOWN", "CLOSED_CLAIMED", "OPEN_CLAIMED")
        ):
            raise PersonalReleaseReconciliationError("Malformed claimed v2 installation state")


@dataclass(frozen=True, slots=True)
class PersonalReleaseReconciliation:
    """A decision about the internal consistency of untrusted evidence ONLY."""

    decision: str
    findings: tuple[str, ...]
    source_git_blob: str
    candidate_sha256: str
    release_id_claimed: str | None
    evidence_digest: str | None

    def manifest(self) -> dict[str, object]:
        return {
            "schema": PERSONAL_RELEASE_RECONCILIATION_SCHEMA,
            "decision": self.decision,
            "findings": list(self.findings),
            "source_git_blob_lf_normalized": self.source_git_blob,
            "candidate_sha256_proposed": self.candidate_sha256,
            "release_directory_id_claimed": self.release_id_claimed,
            "release_entry_bytes_evidence_sha256": self.evidence_digest,
            "release_evidence_provenance": "CALLER_SUPPLIED_UNTRUSTED",
            "full_release_inventory_verified": False,
            "active_pointer_verified": False,
            "managed_distribution_verified": False,
            "installed_state_verified": False,
            "release_manifest_verified": False,
            "trusted_paths_resolved": False,
            "rollback_snapshot": "NOT_CREATED",
            "release_v2_change_required": True,
            "safe_to_install": False,
            "authority": "NO_F1_AUTHORITY",
            "installation": "NOT_INSTALLED",
            "operator_install_consent": "REQUIRED_SEPARATELY",
            "effective_model": "UNKNOWN",
            "effective_model_observed": False,
            "recovery_completed_on_host": False,
            "proposed_action": (
                "DESIGN_NEW_VERSIONED_RELEASE_AND_REAL_PREFLIGHT"
                if self.decision == "CLAIMS_CONSISTENT_NEW_RELEASE_REQUIRED"
                else "STOP_AND_RECONCILE_UNTRUSTED_CLAIMS"

    def __bool__(self) -> bool:
        raise TypeError("Offline reconciliation is never install permission")


def reconcile_personal_chairman_release(
    plan: PersonalChairmanTransitionPlan,
    journal: PersonalChairmanRecoveryJournal,
    *,
    claimed_v2: ClaimedV2Installation | None = None,
    claimed_release_chairman_bytes: bytes | None = None,
) -> PersonalReleaseReconciliation:
    """Assess only internal consistency of supplied claims, never physical release-v2.

    The full v2 manifest and any real release pointer remain UNREAD and UNVERIFIED.
    The caller's projected Chairman entry cannot prove integrity of the other files.
    """
    if type(plan) is not PersonalChairmanTransitionPlan:
        raise PersonalReleaseReconciliationError("Exact PT2A plan is required")
    if type(journal) is not PersonalChairmanRecoveryJournal:
        raise PersonalReleaseReconciliationError("Exact PT2B journal is required")
    if claimed_v2 is not None and type(claimed_v2) is not ClaimedV2Installation:
        raise PersonalReleaseReconciliationError("Exact claimed release-v2 projection required")
    if claimed_release_chairman_bytes is not None and (
        type(claimed_release_chairman_bytes) is not bytes
        or not 0 < len(claimed_release_chairman_bytes) <= _MAX_CHAIRMAN_BYTES
    ):
        raise PersonalReleaseReconciliationError("Bounded exact claimed release bytes required")

    p = plan.manifest()
    findings: list[str] = []
    if p["reconciliation"] != "CALLER_SUPPLIED_SOURCE_MATCH":
        findings.append("PT2A_BEFORE_COPIES_NOT_SOURCE_ALIGNED")
    before = (p["managed_before_sha256_claimed"], p["distribution_before_sha256_claimed"])
    try:
        stage, _ = replay_personal_chairman_recovery(journal)
    except (PersonalChairmanRecoveryJournalError, ValueError):
        stage = "INVALID"
    if (
        journal.source_git_blob != p["source_git_blob_lf_normalized"]
        or journal.source_sha256 != p["source_sha256_as_supplied"]
        or journal.candidate_sha256 != p["candidate_sha256"]
        or journal.before != before
        or journal.plan_reconciliation != p["reconciliation"]
    ):
        findings.append("PT2B_PLAN_IDENTITY_MISMATCH")
    if stage != "DRAFT" or journal.entries or journal.blockers:
        findings.append("PT2B_RECOVERY_STATE_NOT_FRESH_UNBLOCKED_DRAFT")

    evidence_digest: str | None = None
    if claimed_v2 is None:
        findings.append("RELEASE_V2_PROJECTION_NOT_SUPPLIED")
    else:
        v = claimed_v2
        m = v.manifest
        if (
            m.format != V2_MANIFEST_FORMAT or m.version != 2
            or m.source_origin != "git-objects" or m.object_format != "sha1"
            or m.chairman_path != CHAIRMAN_RELEASE_PATH
            or m.chairman_mode not in ("100644", "100755")
        ):
            findings.append("RELEASE_V2_FORMAT_OR_ENTRY_UNSUPPORTED")
        if v.active_pointer_release is None:
            findings.append("ACTIVE_POINTER_NOT_SUPPLIED")
        elif v.active_pointer_release != v.release_directory_id:
            findings.append("ACTIVE_RELEASE_DIRECTORY_MISMATCH")
        if v.pending_install != "ABSENT_CLAIMED":
            findings.append("PENDING_INSTALL_NOT_CLAIMED_ABSENT")
        if v.plugin_tree_coherence != "MATCH_CLAIMED":
            findings.append("MANAGED_DISTRIBUTION_TREE_UNRECONCILED")
        if v.reparse_points != "NONE_CLAIMED":
            findings.append("TARGET_REPARSE_STATE_UNRESOLVED")
        if v.claude_session != "CLOSED_CLAIMED":
            findings.append("CLAUDE_SESSION_NOT_CLAIMED_CLOSED")
        if (v.managed_before_sha256, v.distribution_before_sha256) != before:
            findings.append("RELEASE_TARGET_DIGEST_CLAIMS_MISMATCH_PT2A")
        if (m.chairman_sha256 != v.distribution_before_sha256
                or m.chairman_sha256 != v.managed_before_sha256):
            findings.append("V2_MANIFEST_VS_TARGET_CLAIMS_MISMATCH")
        if claimed_release_chairman_bytes is None:
            findings.append("RELEASE_CHAIRMAN_BYTES_NOT_SUPPLIED")
        else:
            raw = claimed_release_chairman_bytes
            evidence_digest = _sha256(raw)
            if (
                m.chairman_sha256 != evidence_digest
                or m.chairman_git_blob != _git_blob(raw)
                or m.chairman_bytes != len(raw)
            ):
                findings.append("RELEASE_ENTRY_CONTENT_CLAIM_MISMATCH")
        if m.chairman_sha256 == p["candidate_sha256"]:
            findings.append("CANDIDATE_FALSELY_PRESENTED_AS_EXISTING_V2_ENTRY")

    decision = "STOP_UNRESOLVED" if findings else "CLAIMS_CONSISTENT_NEW_RELEASE_REQUIRED"
    return PersonalReleaseReconciliation(
        decision=decision,
        findings=tuple(findings),
        source_git_blob=p["source_git_blob_lf_normalized"],
        candidate_sha256=p["candidate_sha256"],
        release_id_claimed=claimed_v2.release_directory_id if claimed_v2 else None,
        evidence_digest=evidence_digest,
    )

_HEX = frozenset("0123456789abcdef")

class PersonalCandidateProposalError(ValueError):
    """Refuse unbounded, contradictory or incorrectly scoped proposal inputs."""


def _hex(value: object, length: int) -> bool:
    return type(value) is str and len(value) == length and set(value) <= _HEX


def _release_id(value: object) -> bool:
    return (
        type(value) is str
        and 1 <= len(value) <= 80
        and value[0].isascii()
        and value[0].isalnum()
        and all(char.isascii() and (char.isalnum() or char in "_-") for char in value)
    )


def _findings(value: object) -> bool:
    return (
        type(value) is tuple
        and len(value) <= _MAX_FINDINGS
        and all(
            type(item) is str
            and 1 <= len(item) <= 128
            and item.isascii()
            and all(c.isupper() or c.isdigit() or c == "_" for c in item)
            for item in value
        )
    )


@dataclass(frozen=True, slots=True)
class PersonalChairmanReleaseCandidate:
    """Non-authoritative, content-bound OFFLINE proposal, not a release artifact."""

    receipt_sha256: str
    prior_release_id_claimed: str
    proposed_release_id: str
    source_git_blob_lf: str
    source_sha256_as_supplied: str
    candidate_sha256: str
    candidate_size: int
    unchanged_body_sha256: str
    managed_before_sha256_claimed: str
    distribution_before_sha256_claimed: str
    pt2b_journal_id: str
    pt2c_evidence_sha256_claimed: str | None
    pt2c_decision: str
    findings: tuple[str, ...]

    def __post_init__(self) -> None:
        if (
            not _hex(self.receipt_sha256, 64)
            or not _release_id(self.prior_release_id_claimed)
            or not _release_id(self.proposed_release_id)
            or self.proposed_release_id == self.prior_release_id_claimed
            or not _hex(self.source_git_blob_lf, 40)
            or not _hex(self.source_sha256_as_supplied, 64)
            or not _hex(self.candidate_sha256, 64)
            or type(self.candidate_size) is not int
            or not 0 < self.candidate_size <= 512 * 1024
            or not _hex(self.unchanged_body_sha256, 64)
            or not _hex(self.managed_before_sha256_claimed, 64)
            or not _hex(self.distribution_before_sha256_claimed, 64)
            or not _hex(self.pt2b_journal_id, 64)
            or (self.pt2c_evidence_sha256_claimed is not None
                and not _hex(self.pt2c_evidence_sha256_claimed, 64))
            or self.pt2c_decision not in (
                "CLAIMS_CONSISTENT_NEW_RELEASE_REQUIRED", "STOP_UNRESOLVED"

            or not _findings(self.findings)

            "new_release_source_commit_verified": False,

def propose_personal_chairman_versioned_release(
    plan: PersonalChairmanTransitionPlan,
    journal: PersonalChairmanRecoveryJournal,
    assessment: PersonalReleaseReconciliation,
    *,
    proposed_release_id: str,
) -> PersonalChairmanReleaseCandidate:
    """Bind a release-named draft to immutable-seeming caller claims; NO I/O.

    The receipt is NOT a Git source pin, not a release-v2 manifest, not trusted
    host evidence and not operator consent. Re-evaluate real evidence separately.
    """
    if type(plan) is not PersonalChairmanTransitionPlan:
        raise PersonalCandidateProposalError("Exact PT2-A plan required")
    if type(journal) is not PersonalChairmanRecoveryJournal:
        raise PersonalCandidateProposalError("Exact PT2-B journal required")
    if type(assessment) is not PersonalReleaseReconciliation:
        raise PersonalCandidateProposalError("Exact PT2-C assessment required")
    if not _release_id(proposed_release_id):
        raise PersonalCandidateProposalError("Invalid proposed release identifier")
    if not _findings(assessment.findings):
        raise PersonalCandidateProposalError("Invalid or excessive PT2-C findings")
    if assessment.decision not in (
        "CLAIMS_CONSISTENT_NEW_RELEASE_REQUIRED", "STOP_UNRESOLVED"
    ):
        raise PersonalCandidateProposalError("Invalid PT2-C decision vocabulary")
    if assessment.release_id_claimed is None:
        raise PersonalCandidateProposalError("A claimed prior release identifier is required")
    if not _release_id(assessment.release_id_claimed):
        raise PersonalCandidateProposalError("Invalid prior release identifier")
    if proposed_release_id == assessment.release_id_claimed:
        raise PersonalCandidateProposalError("A new versioned release must use a distinct ID")
    if assessment.evidence_digest is not None and not _hex(assessment.evidence_digest, 64):
        raise PersonalCandidateProposalError("Invalid claimed PT2-C evidence digest")

    m = plan.manifest()
    prior = (m["managed_before_sha256_claimed"], m["distribution_before_sha256_claimed"])
    if not all(_hex(digest, 64) for digest in prior):
        raise PersonalCandidateProposalError("Two claimed before-target hashes required")
    if type(journal.plan_id) is not str or not _hex(journal.plan_id, 64):
        raise PersonalCandidateProposalError("Invalid PT2-B plan fingerprint")

    issues = list(assessment.findings)
    try:
        stage, _ = replay_personal_chairman_recovery(journal)
    except (PersonalChairmanRecoveryJournalError, ValueError):
        stage = "INVALID"
    if (
        journal.before != prior
        or journal.source_git_blob != m["source_git_blob_lf_normalized"]
        or journal.source_sha256 != m["source_sha256_as_supplied"]
        or journal.candidate_sha256 != m["candidate_sha256"]
        or journal.plan_reconciliation != m["reconciliation"]
    ):
        issues.append("PT2B_SOURCE_BINDING_MISMATCH")
    if stage != "DRAFT" or journal.entries or journal.blockers:
        issues.append("PT2B_JOURNAL_NOT_FRESH_UNBLOCKED")
    if m["reconciliation"] != "CALLER_SUPPLIED_SOURCE_MATCH":
        issues.append("PT2A_PRIOR_SOURCE_UNRECONCILED")
    if assessment.source_git_blob != m["source_git_blob_lf_normalized"]:
        issues.append("PT2C_SOURCE_GIT_BLOB_MISMATCH")
    if assessment.candidate_sha256 != m["candidate_sha256"]:
        issues.append("PT2C_CANDIDATE_HASH_MISMATCH")
    if assessment.decision != "CLAIMS_CONSISTENT_NEW_RELEASE_REQUIRED" or assessment.findings:
        issues.append("PT2C_RELEASE_CLAIMS_UNRESOLVED")
    if not _hex(m["candidate_sha256"], 64) or not _hex(m["unchanged_body_sha256"], 64):
        raise PersonalCandidateProposalError("Invalid accepted candidate identity")

    # Sorting and deduplication keep a stable audit regardless of report order;
    # no finding is ever interpreted as permission to continue.
    findings = tuple(sorted(set(issues)))
    if not _findings(findings):
        raise PersonalCandidateProposalError("Excessive candidate blocking findings")
    fields = {
        "schema": PERSONAL_CANDIDATE_PROPOSAL_SCHEMA,
        "prior": assessment.release_id_claimed,
        "new": proposed_release_id,
        "source_blob": m["source_git_blob_lf_normalized"],
        "source_as_supplied": m["source_sha256_as_supplied"],
        "candidate": m["candidate_sha256"],
        "candidate_bytes": m["candidate_bytes"],
        "body": m["unchanged_body_sha256"],
        "prior_targets": prior,
        "journal_id": journal.plan_id,
        "pt2c_decision": assessment.decision,
        "pt2c_claim_evidence": assessment.evidence_digest,
        "findings": findings,
    }
    receipt = hashlib.sha256(
        json.dumps(fields, sort_keys=True, separators=(",", ":"), ensure_ascii=True).encode("ascii")
    ).hexdigest()
    return PersonalChairmanReleaseCandidate(
        receipt_sha256=receipt,
        prior_release_id_claimed=assessment.release_id_claimed,
        proposed_release_id=proposed_release_id,
        source_git_blob_lf=m["source_git_blob_lf_normalized"],
        source_sha256_as_supplied=m["source_sha256_as_supplied"],
        candidate_sha256=m["candidate_sha256"],
        candidate_size=m["candidate_bytes"],
        unchanged_body_sha256=m["unchanged_body_sha256"],
        managed_before_sha256_claimed=prior[0],
        distribution_before_sha256_claimed=prior[1],
        pt2b_journal_id=journal.plan_id,
        pt2c_evidence_sha256_claimed=assessment.evidence_digest,
        pt2c_decision=assessment.decision,
        findings=findings,
    )

class ChairmanPolicyError(ValueError):
    """An invalid policy or template cannot be silently replaced by a fallback."""


@dataclass(frozen=True, slots=True)
class ChairmanPolicy:
    """An immutable policy SNAPSHOT, without any automatic promotion authority."""

    version: int
    family: str
    effort: str

    def __post_init__(self) -> None:
        if type(self.version) is not int or not 1 <= self.version <= 1_000_000:
            raise ChairmanPolicyError("Chairman policy version must be a positive bounded integer")
        if type(self.family) is not str or self.family not in _ALLOWED_FAMILIES:
            raise ChairmanPolicyError(
                "Chairman model family must be an explicit supported family alias"
            )
        if type(self.effort) is not str or self.effort not in _ALLOWED_EFFORTS:
            raise ChairmanPolicyError("Chairman effort must be explicitly supported")

    def to_document(self) -> bytes:
        """Deterministic canonical snapshot, not a persisted or approved activation."""
        values = {
            "schema": POLICY_SCHEMA,
            "version": self.version,
            "family": self.family,
            "effort": self.effort,
        }
        return json.dumps(values, sort_keys=True, separators=(",", ":"), ensure_ascii=True).encode("ascii")

    @classmethod
    def from_document(cls, raw: bytes) -> ChairmanPolicy:
        if type(raw) is not bytes or len(raw) > _MAX_POLICY_BYTES:
            raise ChairmanPolicyError("Invalid bounded Chairman policy document")

        def no_duplicate_keys(pairs: list[tuple[str, Any]]) -> dict[str, Any]:
            value: dict[str, Any] = {}
            for key, item in pairs:
                if key in value:
                    raise ChairmanPolicyError("Duplicate Chairman policy field")
                value[key] = item
            return value

        try:
            values = json.loads(raw.decode("utf-8"), object_pairs_hook=no_duplicate_keys)
        except (ValueError, RecursionError):
            raise ChairmanPolicyError("Invalid Chairman policy JSON") from None
        if not isinstance(values, dict) or set(values) != {"schema", "version", "family", "effort"}:
            raise ChairmanPolicyError("Chairman policy fields are incomplete or unknown")
        if values["schema"] != POLICY_SCHEMA:
            raise ChairmanPolicyError("Unsupported Chairman policy document schema")
        return cls(version=values["version"], family=values["family"], effort=values["effort"])

@dataclass(frozen=True, slots=True)
class ChairmanMaterialization:
    """Immutable candidate derived from its ORIGINAL template, never a host proof.

    Keep the template bytes rather than accepting caller-supplied digest fields.
    Both digests are computed from actual bytes, and constructing a candidate
    with mismatched rendered content is refused.
    """

    template: bytes
    content: bytes
    policy: ChairmanPolicy

    def __post_init__(self) -> None:
        if type(self.content) is not bytes:
            raise ChairmanPolicyError("Chairman candidate content must be bytes")
        if self.content != _render_chairman(self.template, self.policy):
            raise ChairmanPolicyError("Chairman candidate does not match its template and policy")

    @property
    def template_sha256(self) -> str:
        return hashlib.sha256(self.template).hexdigest()

    @property
    def rendered_sha256(self) -> str:
        return hashlib.sha256(self.content).hexdigest()

    def manifest(self) -> dict[str, str | int]:
        return {
            "schema": MATERIALIZATION_SCHEMA,
            "policy_schema": POLICY_SCHEMA,
            "policy_version": self.policy.version,
            "requested_family": self.policy.family,
            "requested_effort": self.policy.effort,
            "policy_sha256": hashlib.sha256(self.policy.to_document()).hexdigest(),
            "template_sha256": self.template_sha256,
            "rendered_sha256": self.rendered_sha256,
        }


def _render_chairman(template: bytes, policy: ChairmanPolicy) -> bytes:
    """Produce a deterministic candidate using the input's LF or CRLF style.

    Git for Windows commonly checks out an LF Git blob as CRLF. Refusing CRLF
    would make the shipped source unrenderable, while silently converting it
    would lose byte identity. Accept only UNIFORM LF or UNIFORM CRLF and keep
    every existing body byte and the exact newline convention.
    """
    if type(policy) is not ChairmanPolicy:
        raise ChairmanPolicyError("An explicit ChairmanPolicy snapshot is required")
    if type(template) is not bytes or not 0 < len(template) <= _MAX_TEMPLATE_BYTES:
        raise ChairmanPolicyError("Chairman template must be bounded UTF-8 bytes")
    try:
        text = template.decode("utf-8")
    except UnicodeDecodeError:
        raise ChairmanPolicyError("Chairman template must be UTF-8") from None

    if "\x00" in text:
        raise ChairmanPolicyError("Chairman template contains forbidden NUL bytes")
    if "\r\n" in text:
        newline = "\r\n"
        remainder = text.replace("\r\n", "")
        if "\r" in remainder or "\n" in remainder:
            raise ChairmanPolicyError("Chairman template contains mixed line endings")
    else:
        newline = "\n"
        if "\r" in text:
            raise ChairmanPolicyError("Chairman template contains unsupported line endings")

    opening = "---" + newline
    if not text.startswith(opening):
        raise ChairmanPolicyError("Chairman template must have LF or CRLF frontmatter")
    raw_header, sep, body = text[len(opening):].partition(newline + "---" + newline)
    if not sep or not body.strip():
        raise ChairmanPolicyError("Chairman template is missing frontmatter closure or body")

    lines = raw_header.split(newline)
    fields: dict[str, str] = {}
    for line in lines:
        key, separator, value = line.partition(": ")
        if (
            not separator
            or not _FRONTMATTER_KEY.fullmatch(key)
            or not value.strip()
            or key in fields
        ):
            raise ChairmanPolicyError("Chairman template has ambiguous frontmatter")
        fields[key] = value
    if set(fields) != _FRONTMATTER_KEYS or fields["name"] != "chairman":
        raise ChairmanPolicyError("Only the exact Chairman agent template is allowed")
    if fields["model"] not in _ALLOWED_FAMILIES or fields["effort"] not in _ALLOWED_EFFORTS:
        raise ChairmanPolicyError("Chairman template has unsupported model/effort fields")

    materialized_lines = [
        f"{key}: {policy.family if key == 'model' else policy.effort if key == 'effort' else value}"
        for key, value in (line.split(": ", 1) for line in lines)
    ]
    return (
        opening + newline.join(materialized_lines) + newline + "---" + newline + body
